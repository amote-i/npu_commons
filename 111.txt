"""
E2E test for NPU per-token prefill weight version attribution
(--enable-prefill-weight-versions).

Flow:
  1. A fresh prompt reports one span covering the whole prompt under the
     initial weight version.
  2. POST /update_weight_version relabels; a longer prompt reusing the cached
     prefix reports two spans: old version over the reused prefix, new
     version over the freshly computed suffix.

Aligned with upstream test_weight_version_spans.py::test_17 (CUDA).

"""

import unittest

import requests

from sglang.test.ascend.test_ascend_utils import LLAMA_3_2_1B_INSTRUCT_WEIGHTS_PATH
from sglang.test.ci.ci_register import register_npu_ci
from sglang.test.test_utils import (
    DEFAULT_TIMEOUT_FOR_SERVER_LAUNCH,
    DEFAULT_URL_FOR_TEST,
    CustomTestCase,
    popen_launch_server,
    terminate_and_kill_process_tree,
)

register_npu_ci(est_time=120, suite="full-1-npu-a3", nightly=True)


class TestNpuPrefillWeightVersions(CustomTestCase):
    """E2E: --enable-prefill-weight-versions span attribution on NPU.

    [Test Category] RL Weight Versions
    [Test Target] --enable-prefill-weight-versions, POST /update_weight_version
    """

    REQUEST_TIMEOUT = 120

    # Radix matching rounds down to NPU page boundaries (page_size=128), so
    # the shared prefix must exceed one page or P2 never reuses P1's KV.
    SHARED_PREFIX = (
        "You are a meticulous assistant. Follow every instruction to the letter, "
        "explain your reasoning step by step, and never skip an intermediate result. "
        "Here is the background material you must rely on for the whole "
        "conversation. "
    ) * 10 + "The capital of France is Paris. The capital of Germany is Berlin."
    PROMPT_P1 = SHARED_PREFIX
    PROMPT_P2 = SHARED_PREFIX + " The capital of Italy is Rome."

    @classmethod
    def setUpClass(cls):
        cls.model = LLAMA_3_2_1B_INSTRUCT_WEIGHTS_PATH
        cls.base_url = DEFAULT_URL_FOR_TEST
        cls.process = popen_launch_server(
            cls.model,
            cls.base_url,
            timeout=DEFAULT_TIMEOUT_FOR_SERVER_LAUNCH,
            other_args=[
                "--attention-backend",
                "ascend",
                "--enable-prefill-weight-versions",
            ],
        )

    @classmethod
    def tearDownClass(cls):
        # SIGTERM first so the server releases NPU memory itself.
        terminate_and_kill_process_tree(cls.process)

    def _generate(self, text: str, max_new_tokens: int = 8):
        """Send a deterministic /generate request, return meta_info."""
        resp = requests.post(
            self.base_url + "/generate",
            json={
                "text": text,
                "sampling_params": {
                    "temperature": 0,
                    "max_new_tokens": max_new_tokens,
                    "ignore_eos": True,
                },
            },
            timeout=self.REQUEST_TIMEOUT,
        )
        resp.raise_for_status()
        return resp.json()["meta_info"]

    def _set_weight_version(self, new_version: str):
        """Relabel; abort_all_requests=False keeps the radix cache intact."""
        resp = requests.post(
            self.base_url + "/update_weight_version",
            json={
                "new_version": new_version,
                "abort_all_requests": False,
            },
            timeout=30,
        )
        resp.raise_for_status()

    def _flush_cache(self):
        requests.post(self.base_url + "/flush_cache", timeout=30).raise_for_status()

    def test_prefill_weight_versions(self):
        """
        S1: Fresh prompt → single span under the initial version.
        S2: Version relabel + prefix reuse → split spans with correct
            attribution on each side of the cache-hit boundary.
        """
        # CI retries re-run only this method on the same server — restore
        # fresh-server state first, or S1 hits the previous attempt's cache.
        self._flush_cache()
        self._set_weight_version("default")

        # ── S1: fresh prompt, whole prompt computed under "default" ──
        meta1 = self._generate(self.PROMPT_P1)
        spans1 = meta1["prefill_weight_versions"]
        print(f"[DEBUG] S1 prompt_tokens={meta1['prompt_tokens']}, spans={spans1}")
        self.assertEqual(
            spans1,
            [{"version": "default", "start": 0, "end": meta1["prompt_tokens"]}],
            f"S1: fresh prompt should be a single span, got {spans1}",
        )

        # ── S2: relabel, then reuse P1's cached prefix inside P2 ──
        self._set_weight_version("v2")

        meta2 = self._generate(self.PROMPT_P2)
        spans2 = meta2["prefill_weight_versions"]
        print(
            f"[DEBUG] S2 cached_tokens={meta2['cached_tokens']}, "
            f"prompt_tokens={meta2['prompt_tokens']}, spans={spans2}"
        )
        # cached_tokens (page-aligned prefix-match length) is the split point:
        # reused "default" KV below it, freshly computed "v2" suffix above.
        cached = meta2["cached_tokens"]
        self.assertEqual(
            spans2,
            [
                {"version": "default", "start": 0, "end": cached},
                {
                    "version": "v2",
                    "start": cached,
                    "end": meta2["prompt_tokens"],
                },
            ],
            "S2: expected a stale 'default' span over the reused prefix "
            f"(cached_tokens={cached}) followed by a 'v2' span over the rest",
        )
        # Output-side attribution rides along in the same response
        self.assertEqual(meta2["weight_version"], "v2", "S2: active version")
        self.assertEqual(
            [span["version"] for span in meta2["weight_versions"]],
            ["v2"],
            f"S2: output tokens all generated under v2, got {meta2['weight_versions']}",
        )


if __name__ == "__main__":
    unittest.main()
